using Tallow;

namespace {NAMESPACE};

/// <summary>
/// Code-behind for UI/sample_menu.tui. Put this and a UI Document on the same object. Methods named in
/// onclick="..." / onload="..." are called by the document; fields named in bind="..." are shown (and, for
/// inputs, edited) by it.
/// </summary>
public class SampleMenu : Script
{
    public string playerName = "Player";
    public float volume = 0.6f;

    private UIDocument Document => GetComponent<UIDocument>();

    private void SetStatus(string text) => Document.Find("status")?.SetText(text);

    public void OnOpen() => SetStatus("Menu loaded.");

    public void OnPlay() => SetStatus($"Have fun, {playerName}!");

    public void OnLouder() => volume = Math.Clamp(volume + 0.1f, 0f, 1f);
    public void OnQuieter() => volume = Math.Clamp(volume - 0.1f, 0f, 1f);

    /// <summary>A handler may take the element that was pressed.</summary>
    public void OnClose(UIElement button)
    {
        if (Document.Find("screen") is { } screen)
            screen.Visible = false;
    }
}
