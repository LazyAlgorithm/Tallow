using Tallow;

namespace {NAMESPACE};

/// <summary>Example in-game interface drawn with the GUI class.</summary>
public class Hud : Script
{
    private bool _showHelp = true;

    public override void OnGUI()
    {
        GUI.Box(12, 12, 250, _showHelp ? 118 : 34, new Color(0, 0, 0, 0.45f));
        GUI.Label(22, 20, $"{(int)Time.SmoothedFps} fps");
        if (GUI.Button(170, 16, 82, 24, _showHelp ? "Hide help" : "Help"))
            _showHelp = !_showHelp;
        if (_showHelp)
        {
            GUI.Label(22, 48, "WASD move, Space jump");
            GUI.Label(22, 66, "Hold right mouse to look");
            GUI.Label(22, 84, "Left mouse digs, middle builds");
            GUI.Label(22, 102, "Dig through the boulder to break it");
        }
        if (Scene.IsNetworked)
            GUI.Label(GUI.Width - 220, 20, $"{(Scene.IsServer ? "Host" : "Client")}  tick {Scene.NetworkTick}");
    }
}
