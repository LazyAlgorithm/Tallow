using System;
using Tallow;

namespace {NAMESPACE};

/// Picks the animation for a character from how it is moving: Idle, Walk, Run or Jump. Put it on the character
/// (the Animator can be on a child, the visual model). It only looks at how far the object moved each update, so it
/// works the same whether a Character Controller, a Rigidbody or the network moves it: for the local player and for
/// other players alike.
[Component]
class CharacterAnimation : Script
{
	public float walkSpeed = 0.4f;
	public float runSpeed = 4.5f;
	public float airSpeed = 1.6f;

	Animator mAnimator;
	Vector3 mLastPosition;
	bool mMoved;
	float mAirTime;

	public override void Start()
	{
		mAnimator = GameObject.GetComponentInChildren<Animator>();
		mLastPosition = Transform.Position;
	}

	public override void Update(TickInput input, float deltaTime)
	{
		if (!mAnimator.IsValid || deltaTime <= 0)
			return;
		let position = Transform.Position;
		let velocity = (position - mLastPosition) / deltaTime;
		mLastPosition = position;
		// The first update has nothing to compare with.
		if (!mMoved)
		{
			mMoved = true;
			return;
		}
		let ground = Math.Sqrt(velocity.x * velocity.x + velocity.z * velocity.z);
		// A short delay stops slopes and steps from flickering into the jump pose.
		mAirTime = Math.Abs(velocity.y) > airSpeed ? mAirTime + deltaTime : 0;
		if (mAirTime > 0.12f)
			mAnimator.Play("Jump", 0.1f);
		else if (ground > runSpeed)
			mAnimator.Play("Run", 0.18f);
		else if (ground > walkSpeed)
			mAnimator.Play("Walk", 0.18f);
		else
			mAnimator.Play("Idle", 0.25f);
		// Match the stride to the actual speed so feet do not slide.
		let clip = mAnimator.CurrentClip;
		if (clip == "Walk")
			mAnimator.speed = Mathf.Clamp(ground / 2.6f, 0.6f, 1.6f);
		else if (clip == "Run")
			mAnimator.speed = Mathf.Clamp(ground / 6.0f, 0.8f, 1.4f);
		else
			mAnimator.speed = 1.0f;
	}
}
