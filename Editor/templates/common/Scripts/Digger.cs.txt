using Tallow;

namespace {NAMESPACE};

/// <summary>
/// Left click digs, middle click builds, aimed with the camera it is on. Works on voxel terrain and on destructible
/// VoxelObjects (which break apart when cut through).
/// </summary>
public class Digger : Script
{
    public float Radius = 3.0f;
    public float Strength = 1.5f;

    // A local camera tool: this callback is not replayed as predicted simulation.
    public override void LocalUpdate(float deltaTime)
    {
        bool dig = Input.GetMouseButton(MouseButton.Left);
        bool build = Input.GetMouseButton(MouseButton.Middle);
        if ((!dig && !build) || GUI.MouseOverControl)
            return;
        if (Scene.RaycastHit(Transform.Position, Transform.Forward, 200) is not { } hit)
            return;
        if (hit.HitObject?.GetComponent<VoxelObject>() is { IsValid: true } voxels)
        {
            if (dig)
                voxels.Carve(hit.Point, Radius * 0.3f);
            return;
        }
        VoxelTerrain.Find(Scene)?.ModifySphere(hit.Point, Radius, (dig ? -Strength : Strength) * Time.DeltaTime * 4);
    }
}
