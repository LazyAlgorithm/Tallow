using System.Numerics;
using Tallow;

namespace {NAMESPACE};

/// <summary>Example script: rotates its object. Public fields show up in the Inspector.</summary>
public class Spinner : Script
{
    public Vector3 DegreesPerSecond = new(0, 45, 0);

    public override void Update(TickInput input, float deltaTime) =>
        Transform.Rotation *= Quaternion.Euler(DegreesPerSecond.X * deltaTime, DegreesPerSecond.Y * deltaTime, DegreesPerSecond.Z * deltaTime);
}
